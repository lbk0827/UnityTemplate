**sf에서 추출한 규칙(권위):**
- 충전: 매 틱 `intervals = floor((now-anchor)/interval)`; `cur = min(cur + intervals*amount, max)`; `anchor += intervals*interval`(분수 진행 보존); `cur>=max → anchor=null`. 오프라인 복귀는 같은 식으로 한 번에 적용.
- 앵커: 값 세팅 시 `new>=max → null`; `max 이상→미만 전이 → now`; 그 외 유지(4→3 소비는 진행 중 사이클 유지). 로드 시 `anchor==null && cur<max → now`.
- 표시: 다음 하트까지 `interval - (now-anchor)`(0 클램프), 가득 차면 FULL. 풀 충전 시각 `anchor + ceil((max-cur)/amount)*interval`.
- 무한 버프: 버프 재화의 값은 남은 초; `Set/Add`는 `end = now + newValue`; `Set(0)` 해제. 하트의 `InfiniteEnabled = max(버프 end) > now`. 버프 중엔 차감/환불 모두 생략, `CanUseEntryCost = true`.
- 입장 게이트: 시도당 정확히 1회 차감(`StartStage` 단일 관문). `ShouldCharge(stage) = stage > EarlyLevelCutoffStage`. 클리어 시 같은 조건으로 +1 환불(최대치 이상이면 환불 생략, 무한 버프면 생략). 실패/컨티뉴는 변화 없음. 재시도는 게이트를 다시 탄다. 면제는 차감·환불 대칭.
- 클리어 시 `CurrentStage = next`를 **즉시 저장**(팝업 전).
- PendingRewardQueue: 메모리 큐, `PendingReward{CurrencyId, Amount, From, To, AlreadyCredited}`. 보상은 즉시 지급 + `CurrencyDisplayLock.Engage(id, from)`로 HUD는 옛 값을 보여주고, 코인 도착 시 `Release`.
- 컨티뉴 가격: `min(base + used*add, max)`; `CanAfford = gold>=price`; 이동/시간 중 이동 우선; 사유→오퍼 매핑은 TimeOver/MovesZero만; `usedCount`는 시도 시작마다 0.
- 프리로더 레지스트리: 마지막 등록 우선, 연속 StartPreload는 이전 취소, 세대 id로 `IsPreloading` 정합, OCE는 로그 없음.

`RechargeLogic.cs` (순수):

```csharp
using System;

namespace BK.Meta
{
    /// <summary>Rechargeable currency math. State in, state out; no clocks, no Unity.</summary>
    public static class RechargeLogic
    {
        public readonly struct State
        {
            public readonly long Value;
            /// <summary>Start of the cycle currently in progress. Null when full.</summary>
            public readonly DateTime? Anchor;
            public State(long value, DateTime? anchor) { Value = value; Anchor = anchor; }
        }

        /// <summary>Applies every whole interval elapsed since the anchor (also the offline catch-up).</summary>
        public static State Advance(State s, CurrencyDefinition d, DateTime now)
        {
            if (d.Kind != CurrencyKind.Rechargeable || s.Value >= d.RechargeMax || s.Anchor == null
                || d.RechargeInterval <= TimeSpan.Zero || d.RechargeAmount <= 0)
                return s;
            var intervals = (long)Math.Floor((now - s.Anchor.Value).TotalSeconds / d.RechargeInterval.TotalSeconds);
            if (intervals <= 0) return s;
            var value = Math.Min(s.Value + intervals * d.RechargeAmount, d.RechargeMax);
            DateTime? anchor = s.Anchor.Value + TimeSpan.FromTicks(d.RechargeInterval.Ticks * intervals);
            if (value >= d.RechargeMax) anchor = null;
            return new State(value, anchor);
        }

        /// <summary>Anchor rules when the value is set directly (spend, refund, purchase).</summary>
        public static State SetValue(State s, CurrencyDefinition d, long newValue, DateTime now)
        {
            if (d.Kind != CurrencyKind.Rechargeable) return new State(newValue, null);
            if (newValue >= d.RechargeMax) return new State(newValue, null);
            if (s.Value >= d.RechargeMax) return new State(newValue, now);      // crossed from full to not full
            return new State(newValue, s.Anchor ?? now);                        // keep the running cycle
        }

        /// <summary>Normalises a loaded state: a non-full value always has an anchor.</summary>
        public static State OnLoad(State s, CurrencyDefinition d, DateTime now)
            => d.Kind == CurrencyKind.Rechargeable && s.Value < d.RechargeMax && s.Anchor == null ? new State(s.Value, now) : s;

        public static TimeSpan TimeToNext(State s, CurrencyDefinition d, DateTime now)
        {
            if (s.Value >= d.RechargeMax || s.Anchor == null) return TimeSpan.Zero;
            var remaining = d.RechargeInterval - (now - s.Anchor.Value);
            return remaining < TimeSpan.Zero ? TimeSpan.Zero : remaining;
        }

        public static DateTime? FullAt(State s, CurrencyDefinition d)
        {
            if (s.Value >= d.RechargeMax || s.Anchor == null || d.RechargeAmount <= 0) return null;
            var cycles = (long)Math.Ceiling((d.RechargeMax - s.Value) / (double)d.RechargeAmount);
            return s.Anchor.Value + TimeSpan.FromTicks(d.RechargeInterval.Ticks * cycles);
        }
    }
}
```

`Wallet.cs`:

```csharp
using System;
using System.Collections.Generic;
using R3;
using BK.Core.Time;
using BK.Save;

namespace BK.Meta
{
    /// <inheritdoc cref="IWallet"/>
    public sealed class Wallet : IWallet, IDisposable
    {
        private readonly Dictionary<string, Currency> _currencies = new();
        private readonly WalletData _data;
        private readonly IClock _clock;
        private readonly Subject<(string, long, string)> _changed = new();

        public Wallet(ICurrencyCatalog catalog, ISaveService saves, IClock clock)
        {
            _clock = clock;
            _data = saves.Get<WalletData>();
            var now = clock.UtcNow;
            foreach (var d in catalog.Definitions)
            {
                var entry = _data.Find(d.Id);
                if (entry == null) { entry = new WalletData.Entry { id = d.Id, value = d.InitialValue }; _data.entries.Add(entry); _data.MarkDirty(); }
                DateTime? anchor = entry.anchorTicks == 0 ? null : new DateTime(entry.anchorTicks, DateTimeKind.Utc);
                var c = new Currency(d, entry.value, anchor);
                if (d.Kind == CurrencyKind.Buff) { c.BuffEnd = anchor; c.Anchor = null; }
                else if (d.Kind == CurrencyKind.Rechargeable)
                {
                    var s = RechargeLogic.OnLoad(new RechargeLogic.State(entry.value, anchor), d, now);
                    c.Anchor = s.Anchor; Persist(c, s.Value, s.Anchor);
                }
                _currencies.Add(d.Id, c);
            }
            Tick(now);
        }

        public Observable<(string id, long delta, string reason)> Changed => _changed;
        public Currency Get(string id) => _currencies.TryGetValue(id, out var c) ? c : throw new KeyNotFoundException($"currency '{id}' is not in the catalog");
        public long ValueOf(string id) => Get(id)._value.Value;
        public bool CanAfford(string id, long amount) => ValueOf(id) >= amount;
        public bool IsInfinite(string id) => Get(id)._infinite.Value;
        public TimeSpan TimeToNext(string id) { var c = Get(id); return RechargeLogic.TimeToNext(new(c._value.Value, c.Anchor), c.Definition, _clock.UtcNow); }

        public void Add(string id, long delta, string reason) => Set(id, Math.Max(0, ValueOf(id) + delta), reason, delta);
        public void Set(string id, long value, string reason) => Set(id, value, reason, value - ValueOf(id));

        private void Set(string id, long value, string reason, long delta)
        {
            var c = Get(id); var now = _clock.UtcNow;
            if (c.Definition.Kind == CurrencyKind.Buff)
            {
                c.BuffEnd = value > 0 ? now + TimeSpan.FromSeconds(value) : null;
                Persist(c, value, c.BuffEnd);
                RefreshInfinite(now);
            }
            else
            {
                var s = RechargeLogic.SetValue(new(c._value.Value, c.Anchor), c.Definition, value, now);
                c.Anchor = s.Anchor; Persist(c, s.Value, s.Anchor);
            }
            _changed.OnNext((id, delta, reason));
        }

        public void Tick(DateTime now)
        {
            foreach (var c in _currencies.Values)
            {
                if (c.Definition.Kind == CurrencyKind.Buff)
                {
                    var remain = c.BuffEnd.HasValue ? (long)Math.Max(0, (c.BuffEnd.Value - now).TotalSeconds) : 0;
                    if (remain != c._value.Value) { if (remain == 0) c.BuffEnd = null; Persist(c, remain, c.BuffEnd); }
                    continue;
                }
                if (c.Definition.Kind != CurrencyKind.Rechargeable) continue;
                var s = RechargeLogic.Advance(new(c._value.Value, c.Anchor), c.Definition, now);
                if (s.Value != c._value.Value || s.Anchor != c.Anchor) { c.Anchor = s.Anchor; Persist(c, s.Value, s.Anchor); }
            }
            RefreshInfinite(now);
        }

        private void RefreshInfinite(DateTime now)
        {
            foreach (var c in _currencies.Values)
            {
                if (c.Definition.Kind != CurrencyKind.Rechargeable) continue;
                var on = false;
                foreach (var buffId in c.Definition.InfiniteBuffIds)
                    if (_currencies.TryGetValue(buffId, out var b) && b.BuffEnd.HasValue && b.BuffEnd.Value > now) { on = true; break; }
                c._infinite.Value = on;
            }
        }

        private void Persist(Currency c, long value, DateTime? stamp)
        {
            c._value.Value = value;
            var e = _data.Find(c.Definition.Id);
            e.value = value; e.anchorTicks = stamp?.Ticks ?? 0;
            _data.MarkDirty();
        }

        public void Dispose() { foreach (var c in _currencies.Values) c.Dispose(); _changed.Dispose(); }
    }

    /// <summary>Drives recharge/buff expiry once per frame.</summary>
    public sealed class WalletTicker : VContainer.Unity.ITickable
    {
        private readonly IWallet _wallet; private readonly IClock _clock;
        public WalletTicker(IWallet wallet, IClock clock) { _wallet = wallet; _clock = clock; }
        public void Tick() => _wallet.Tick(_clock.UtcNow);
    }
}
```

```csharp
using System;
using R3;
using BK.Save;

namespace BK.Meta
{
    [Serializable] public sealed class StageProgressData : SaveData { public override int CurrentVersion => 1; public int currentStage = 1; public int attempts; public override bool Validate() => currentStage >= 1 && attempts >= 0; }

    public sealed class StageProgress : IStageProgress, IDisposable
    {
        public const string ReasonEntry = "stage_entry";
        public const string ReasonRefund = "stage_refund";
        private readonly StageProgressData _data;
        private readonly IWallet _wallet;
        private readonly StageEntryPolicy _policy;
        private readonly ReactiveProperty<int> _current;
        private readonly Subject<StageStarted> _started = new();
        private readonly Subject<StageCleared> _cleared = new();

        public StageProgress(ISaveService saves, IWallet wallet, StageEntryPolicy policy)
        { _data = saves.Get<StageProgressData>(); _wallet = wallet; _policy = policy; _current = new(_data.currentStage); }

        public ReadOnlyReactiveProperty<int> CurrentStage => _current;
        public int AttemptsOnCurrentStage => _data.attempts;
        public Observable<StageStarted> Started => _started;
        public Observable<StageCleared> Cleared => _cleared;

        public bool CanEnter(int stage) => !_policy.ShouldCharge(stage) || _wallet.IsInfinite(_policy.EnergyId) || _wallet.ValueOf(_policy.EnergyId) > 0;

        public bool TryStart(int stage)
        {
            if (_policy.ShouldCharge(stage) && !_wallet.IsInfinite(_policy.EnergyId))
            {
                if (_wallet.ValueOf(_policy.EnergyId) <= 0) return false;
                _wallet.Add(_policy.EnergyId, -1, ReasonEntry);
            }
            if (stage != _data.currentStage) { _data.currentStage = stage; _data.attempts = 0; _current.Value = stage; }
            _data.attempts++; _data.MarkDirty();
            _started.OnNext(new StageStarted(stage, _data.attempts));
            return true;
        }

        public void Clear(int stage)
        {
            // Saved first: a crash during the clear presentation must not replay the stage.
            if (stage >= _data.currentStage) { _data.currentStage = stage + 1; _data.attempts = 0; _current.Value = _data.currentStage; _data.MarkDirty(); }
            if (_policy.ShouldCharge(stage) && !_wallet.IsInfinite(_policy.EnergyId))
            {
                var c = _wallet.Get(_policy.EnergyId);
                if (c.Value.CurrentValue < c.Definition.RechargeMax) _wallet.Add(_policy.EnergyId, +1, ReasonRefund);
            }
            _cleared.OnNext(new StageCleared(stage));
        }

        public void Dispose() { _current.Dispose(); _started.Dispose(); _cleared.Dispose(); }
    }
}
```

```csharp
namespace BK.Meta
{
    public enum StageFailReason { Unknown, TimeOver, HpZero, MovesZero, UserQuit, GameSpecific }

    /// <summary>One continue product. Supplied by the game.</summary>
    public sealed class ContinueOfferDefinition
    {
        public StageFailReason Reason; public string CurrencyId = "Gold";
        public long BasePrice; public long AddPrice; public long MaxPrice;
        public int AddMoves; public int AddTimeSeconds;
    }
    public interface IContinueOfferCatalog { System.Collections.Generic.IReadOnlyList<ContinueOfferDefinition> Offers { get; } }

    public static class FailContinueLogic
    {
        public static long ComputePrice(long basePrice, long addPrice, long maxPrice, int usedCount) => System.Math.Min(basePrice + usedCount * addPrice, maxPrice);
        public static bool CanAfford(long balance, long price) => balance >= price;
        /// <summary>Moves win over time when both are present.</summary>
        public static (bool isMoves, int amount) ResolveGrant(int addMoves, int addTimeSeconds) => addMoves > 0 ? (true, addMoves) : (false, addTimeSeconds);
        public static bool OffersContinue(StageFailReason r) => r == StageFailReason.TimeOver || r == StageFailReason.MovesZero;
    }

    public readonly struct ContinueOffer
    {
        public readonly ContinueOfferDefinition Definition; public readonly long Price; public readonly bool CanAfford;
        public ContinueOffer(ContinueOfferDefinition d, long price, bool canAfford) { Definition = d; Price = price; CanAfford = canAfford; }
    }

    /// <summary>Per-attempt continue ladder. Resets when a stage attempt starts.</summary>
    public sealed class ContinueOffers : System.IDisposable
    {
        private readonly IContinueOfferCatalog _catalog; private readonly IWallet _wallet; private readonly System.IDisposable _reset;
        public int UsedCount { get; private set; }
        public ContinueOffers(IContinueOfferCatalog catalog, IWallet wallet, IStageProgress progress)
        { _catalog = catalog; _wallet = wallet; _reset = progress.Started.Subscribe(_ => UsedCount = 0); }
        public bool TryGetOffer(StageFailReason reason, out ContinueOffer offer)
        {
            offer = default;
            if (!FailContinueLogic.OffersContinue(reason)) return false;
            foreach (var d in _catalog.Offers)
            {
                if (d.Reason != reason) continue;
                var price = FailContinueLogic.ComputePrice(d.BasePrice, d.AddPrice, d.MaxPrice, UsedCount);
                offer = new ContinueOffer(d, price, FailContinueLogic.CanAfford(_wallet.ValueOf(d.CurrencyId), price));
                return true;
            }
            return false;
        }
        /// <summary>Charges the offer's price; false (no charge) when unaffordable. Increments the ladder on success.</summary>
        public bool TryPurchase(in ContinueOffer offer)
        {
            if (!_wallet.CanAfford(offer.Definition.CurrencyId, offer.Price)) return false;
            _wallet.Add(offer.Definition.CurrencyId, -offer.Price, "continue");
            UsedCount++;
            return true;
        }
        public void Dispose() => _reset.Dispose();
    }
}
```
